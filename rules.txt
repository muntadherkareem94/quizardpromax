rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // --- Helper Functions ---
    function isAdmin() {
      return request.auth != null && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    // --- Discipline Rules ---
    match /disciplines/{disciplineId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // --- User Rules ---
    match /users/{userId} {
      allow create: if isOwner(userId);
      allow read: if isOwner(userId) || isAdmin();
      allow list: if isAdmin();
      allow update: if isOwner(userId);
      allow delete: if false;

      match /quizResults/{resultId} {
        allow read, create: if isOwner(userId);
        allow update, delete: if false;
      }
    }

    // --- User Question History Rules ---
    match /userQuestionHistory/{historyId} {
      allow get: if request.auth.uid == historyId.split('_')[0];
      allow list: if request.auth.uid == resource.data.userId;
      allow write: if request.auth.uid == request.resource.data.userId;
    }

    // --- NEW: Question Statistics Rules ---
    // This collection will be written to by a backend Firebase Function
    // and read by clients to display audience difficulty.
    match /questionStats/{questionId} {
      allow read: if request.auth != null;
      // Write access should be restricted to your backend function.
      allow write: if false;
    }

    // --- Content Rules (Books, Chapters, Questions) ---
    match /books/{bookId} {
      allow read: if request.auth != null;
      allow write: if isAdmin();

      match /chapters/{chapterId} {
        allow read: if request.auth != null;
        allow write: if isAdmin();

        match /questions/{questionId} {
          allow read: if request.auth != null;
          allow write: if isAdmin();
        }
      }
    }

    // --- Collection Group Rules ---
    match /{path=**}/chapters/{chapterId} {
      allow read: if request.auth != null;
    }

    match /{path=**}/questions/{questionId} {
      allow read: if request.auth != null;
    }
  }
}
